local Players = game:GetService("Players")
local HttpService = game:GetService("HttpService")
local lp = Players.LocalPlayer

local key = nil
pcall(function()
    if _G.script_key then key = _G.script_key end
    if getgenv and getgenv().script_key then key = getgenv().script_key end
    if script_key then key = script_key end
end)

if not key or key == "" then
    warn("❌ NO KEY - Fais: script_key = \"ZENO-TEST-123\"")
    return
end

local url = "https://zeno-finder-default-rtdb.europe-west1.firebasedatabase.app/licenses/"..tostring(key)..".json"
local ok, body = pcall(function() return game:HttpGet(url) end)
if not ok or body == "null" or body == "" then
    warn("❌ CLE INVALIDE: "..tostring(key))
    return
end

local ok2, data = pcall(function() return HttpService:JSONDecode(body) end)
if not ok2 or data.active == false then
    warn("🚫 BANNED")
    return
end

_G.ZENO_AUTH = true
_G.ZENO_KEY_VALID = key
getgenv().ZENO_AUTH = true
getgenv().ZENO_KEY_VALID = key

print("✅ AUTH OK - "..tostring(key))

local hubUrl = "https://raw.githubusercontent.com/ZENOHUBPAID/loader.lua/refs/heads/main/HUB_PRIVE_LITE.txt"
local ok3, hubCode = pcall(function() return game:HttpGet(hubUrl) end)
if not ok3 then
    warn("Erreur HttpGet HUB: "..tostring(hubCode))
    return
end

-- Check if 404
if hubCode:sub(1,3) == "404" or hubCode:find("<!DOCTYPE") or hubCode:find("404: Not Found") then
    warn("❌ HUB 404 - Fichier HUB_PRIVE_LITE.txt introuvable sur GitHub")
    return
end

local f, err = loadstring(hubCode)
if f then f() else warn("Erreur HUB: "..tostring(err)) end
