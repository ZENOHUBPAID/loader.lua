-- ZENO ULTRA SIMPLE - TEST DIRECT
local Players = game:GetService("Players")
local HttpService = game:GetService("HttpService")
local lp = Players.LocalPlayer

-- Affiche direct un message pour voir si le script se lance
local gui = Instance.new("ScreenGui", lp:WaitForChild("PlayerGui"))
gui.Name = "ZENO_TEST"
local lbl = Instance.new("TextLabel", gui)
lbl.Size = UDim2.new(0,400,0,100)
lbl.Position = UDim2.new(0.5,-200,0.5,-50)
lbl.BackgroundColor3 = Color3.fromRGB(20,20,20)
lbl.TextColor3 = Color3.fromRGB(0,255,0)
lbl.TextSize = 20
lbl.Text = "ZENO LOADER LANCE !\nVerification..."
Instance.new("UICorner", lbl)

task.wait(1)

local key = nil
pcall(function()
    -- On check TOUTES les variantes que les autres hubs utilisent
    if _G.script_key then key = _G.script_key end
    if _G.ZENO_KEY then key = _G.ZENO_KEY end
    if getgenv then
        if getgenv().script_key then key = getgenv().script_key end
        if getgenv().ZENO_KEY then key = getgenv().ZENO_KEY end
    end
    if _G.getgenv and _G.getgenv().script_key then key = _G.getgenv().script_key end
    -- Variable globale directe (comme script_key = "...")
    if script_key then key = script_key end
    if ZENO_KEY then key = ZENO_KEY end
end)
lbl.Text = "Cle: "..tostring(key).."\nChargement Firebase..."

-- TEST FIREBASE AVEC HttpGet simple
local url = "https://zeno-finder-default-rtdb.europe-west1.firebasedatabase.app/licenses/"..tostring(key)..".json"
local ok, body = pcall(function() return game:HttpGet(url) end)

if not ok then
    lbl.TextColor3 = Color3.fromRGB(255,50,50)
    lbl.Text = "ERREUR HttpGet Firebase:\n"..tostring(body).."\n\nActive Http dans le jeu ou utilise un executor avec Http"
    return
end

lbl.Text = "Firebase reponse:\n"..body:sub(1,150)

task.wait(1.5)

if body == "null" or body == "" then
    lbl.TextColor3 = Color3.fromRGB(255,50,50)
    lbl.Text = "Cle invalide !\n"..tostring(key).."\n\nVa dans Firebase et verifie licenses/"..tostring(key)
    return
end

local ok2, data = pcall(function() return HttpService:JSONDecode(body) end)
if not ok2 then
    lbl.TextColor3 = Color3.fromRGB(255,50,50)
    lbl.Text = "Erreur JSON:\n"..body:sub(1,200)
    return
end

if data.active == false then
    lbl.Text = "Cle bannie !"
    return
end

lbl.TextColor3 = Color3.fromRGB(0,255,100)
lbl.Text = "Acces OK !\nChargement HUB..."

task.wait(1)
gui:Destroy()

-- CHARGE LE HUB
local hubUrl = "https://raw.githubusercontent.com/ZENOHUBPAID/loader.lua/refs/heads/main/HUB_PRIVE.txt"
local ok3, hubCode = pcall(function() return game:HttpGet(hubUrl) end)
if not ok3 then
    warn("Erreur HUB HttpGet: "..tostring(hubCode))
    return
end
print("HUB code len: "..#hubCode)
local f, err = loadstring(hubCode)
if f then f() else warn("Loadstring HUB erreur: "..tostring(err)) end
