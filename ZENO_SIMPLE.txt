-- ZENO FINAL LOADER - PROTEGE + DELTA FIX
local Players = game:GetService("Players")
local HttpService = game:GetService("HttpService")
local lp = Players.LocalPlayer

local function getGuiParent()
    local ok, res = pcall(function()
        if gethui then return gethui() end
        if get_hidden_ui then return get_hidden_ui() end
        if get_hidden_gui then return get_hidden_gui() end
        return game:GetService("CoreGui")
    end)
    if ok and res then return res end
    return lp:WaitForChild("PlayerGui")
end

local parent = getGuiParent()
local gui = Instance.new("ScreenGui")
gui.Name = "ZENO_AUTH_"..math.random(10000,99999)
gui.ResetOnSpawn = false
gui.IgnoreGuiInset = true
gui.Parent = parent

local lbl = Instance.new("TextLabel", gui)
lbl.Size = UDim2.new(0, 420, 0, 260)
lbl.Position = UDim2.new(0.5, -210, 0.5, -130)
lbl.BackgroundColor3 = Color3.fromRGB(15,15,15)
lbl.TextColor3 = Color3.fromRGB(0,255,120)
lbl.TextSize = 15
lbl.TextWrapped = true
lbl.Text = "ZENO HUB - Verification..."
Instance.new("UICorner", lbl).CornerRadius = UDim.new(0,10)

local key = nil
pcall(function()
    if _G.script_key then key = _G.script_key end
    if _G.ZENO_KEY then key = _G.ZENO_KEY end
    if getgenv then
        if getgenv().script_key then key = getgenv().script_key end
        if getgenv().ZENO_KEY then key = getgenv().ZENO_KEY end
    end
    if script_key then key = script_key end
    if ZENO_KEY then key = ZENO_KEY end
end)

if not key or key == "" then
    lbl.TextColor3 = Color3.fromRGB(255,60,60)
    lbl.Text = "❌ AUCUNE CLE !\n\nFais:\nscript_key = \"TA_CLE\"\nloadstring(game:HttpGet(\"...\"))()"
    return
end

lbl.Text = "🔑 Cle: "..tostring(key).."\n\n⏳ Verification Firebase..."
task.wait(0.3)

local url = "https://zeno-finder-default-rtdb.europe-west1.firebasedatabase.app/licenses/"..tostring(key)..".json"
local ok, body = pcall(function()
    if http_request then return http_request({Url=url, Method="GET"}).Body
    elseif request then return request({Url=url, Method="GET"}).Body
    else return game:HttpGet(url) end
end)

if not ok or body == "null" or body == "" then
    lbl.TextColor3 = Color3.fromRGB(255,60,60)
    lbl.Text = "❌ CLE INVALIDE !\n"..tostring(key)
    return
end

local ok2, data = pcall(function() return HttpService:JSONDecode(body) end)
if not ok2 or data.active == false then
    lbl.TextColor3 = Color3.fromRGB(255,60,60)
    lbl.Text = "🚫 CLE BANNIE OU INVALIDE"
    return
end

-- SET AUTH TOKEN - INDISPENSABLE POUR LE HUB PROTEGE
_G.ZENO_AUTH = true
_G.ZENO_KEY_VALID = key
_G.ZENO_SECRET = HttpService:GenerateGUID(false)
_G.ZENO_USER_DATA = data
getgenv().ZENO_AUTH = true
getgenv().ZENO_KEY_VALID = key
getgenv().ZENO_SECRET = _G.ZENO_SECRET

lbl.TextColor3 = Color3.fromRGB(0,255,120)
lbl.Text = "✅ ACCES AUTORISE !\n"..tostring(key).."\n🚀 Chargement HUB..."
task.wait(0.8)
gui:Destroy()

-- CHARGE LE HUB OBFUSQUE
local hubUrl = "https://raw.githubusercontent.com/ZENOHUBPAID/loader.lua/refs/heads/main/HUB_PRIVE_OBFUSCATED.txt"
local ok3, hubCode = pcall(function()
    if http_request then return http_request({Url=hubUrl, Method="GET"}).Body
    elseif request then return request({Url=hubUrl, Method="GET"}).Body
    else return game:HttpGet(hubUrl) end
end)

if not ok3 then
    local g2 = Instance.new("ScreenGui", getGuiParent())
    local l2 = Instance.new("TextLabel", g2)
    l2.Size = UDim2.new(0,400,0,120)
    l2.Position = UDim2.new(0.5,-200,0.5,-60)
    l2.Text = "Erreur HUB: "..tostring(hubCode)
    return
end

local f, err = loadstring(hubCode)
if f then f() else
    local g2 = Instance.new("ScreenGui", getGuiParent())
    local l2 = Instance.new("TextLabel", g2)
    l2.Size = UDim2.new(0,500,0,200)
    l2.Position = UDim2.new(0.5,-250,0.5,-100)
    l2.Text = "Erreur HUB: "..tostring(err)
    l2.BackgroundColor3 = Color3.fromRGB(40,0,0)
    l2.TextColor3 = Color3.fromRGB(255,255,255)
    l2.TextWrapped = true
end
