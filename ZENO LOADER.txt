-- Fonction pour checker la licence dans Firebase
local function httpRequest(url)
    local ok, res = pcall(function()
        if syn and syn.request then
            return syn.request({Url = url, Method = "GET"}).Body
        elseif http and http.request then
            return http.request({Url = url, Method = "GET"}).Body
        elseif request then
            return request({Url = url, Method = "GET"}).Body
        else
            return HttpService:GetAsync(url)
        end
    end)
    if ok then return res end
    return nil
end

local function checkAccess()
    -- 1. Vérif si clé dans getgenv
    local userKey = nil
    pcall(function()
        if getgenv and getgenv().ZENO_KEY then
            userKey = getgenv().ZENO_KEY
        end
        if _G.ZENO_KEY then
            userKey = _G.ZENO_KEY
        end
    end)
    
    if not userKey or userKey == "" then
        -- Pas de clé -> demande UI
        -- Pour l'instant on refuse direct, tu peux ajouter une UI de saisie
        return false, "Aucune clé fournie. Mettez getgenv().ZENO_KEY = \"VOTRE_CLE\" avant le loadstring"
    end
    
    -- 2. Check Firebase /licenses/KEY
    local url = FIREBASE_URL .. "licenses/" .. HttpService:UrlEncode(userKey) .. ".json?auth=" .. FIREBASE_TOKEN
    -- Si tu utilises règles avec token dans data, enlève ?auth et check le contenu
    local simpleUrl = FIREBASE_URL .. "licenses/" .. HttpService:UrlEncode(userKey) .. ".json"
    
    local body = httpRequest(simpleUrl)
    if not body or body == "null" then
        return false, "Clé invalide: " .. tostring(userKey)
    end
    
    local ok, data = pcall(function() return HttpService:JSONDecode(body) end)
    if not ok or not data then
        return false, "Erreur vérification licence"
    end
    
    if data.active == false then
        return false, "Clé bannie"
    end
    
    if data.expire and type(data.expire) == "number" and data.expire < os.time() then
        return false, "Clé expirée"
    end
    
    -- Optionnel: check HWID / UserId whitelist
    if data.hwid and data.hwid ~= "" then
        local currentHwid = ""
        pcall(function()
            currentHwid = tostring(game:GetService("RbxAnalyticsService"):GetClientId())
        end)
        -- Si tu veux bloquer par HWID
        -- if currentHwid ~= data.hwid then return false, "HWID différent" end
    end
    
    if data.userId and data.userId ~= 0 then
        if data.userId ~= player.UserId then
            return false, "Clé liée à un autre compte Roblox"
        end
    end
    
    return true, data
end

-- LANCEMENT
local allowed, info = checkAccess()
if not allowed then
    denyAccess(info)
    return
end

-- Si autorisé, charge le vrai hub
print("[ZENO] Accès autorisé, chargement...")
local ok, hubCode = pcall(function()
    return httpRequest(HUB_URL)
end)

if not ok or not hubCode or #hubCode < 100 then
    warn("[ZENO] Impossible de charger le hub privé")
    return
end

-- Exécute le hub
local func, err = loadstring(hubCode)
if not func then
    warn("[ZENO] Erreur loadstring hub: " .. tostring(err))
    return
end
