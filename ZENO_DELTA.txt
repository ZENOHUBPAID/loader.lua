-- ZENO DELTA FIX - Compatible Delta mobile
local Players = game:GetService("Players")
local HttpService = game:GetService("HttpService")
local lp = Players.LocalPlayer

-- FIX DELTA : utilise gethui() pour contourner le blocage PlayerGui
local function getGuiParent()
    local success, result = pcall(function()
        if gethui then return gethui() end
        if get_hidden_ui then return get_hidden_ui() end
        if get_hidden_gui then return get_hidden_gui() end
        return game:GetService("CoreGui")
    end)
    if success and result then return result end
    return lp:WaitForChild("PlayerGui")
end

local parent = getGuiParent()

local gui = Instance.new("ScreenGui")
gui.Name = "ZENO_DELTA_TEST"
gui.ResetOnSpawn = false
gui.IgnoreGuiInset = true
gui.Parent = parent

local lbl = Instance.new("TextLabel", gui)
lbl.Size = UDim2.new(0, 400, 0, 250)
lbl.Position = UDim2.new(0.5, -200, 0.5, -125)
lbl.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
lbl.TextColor3 = Color3.fromRGB(0, 255, 0)
lbl.TextSize = 16
lbl.TextWrapped = true
lbl.Text = "ZENO LOADER LANCE (DELTA)!\nVerification..."
Instance.new("UICorner", lbl).CornerRadius = UDim.new(0, 8)

local key = nil
pcall(function()
    if _G.script_key then key = _G.script_key end
    if _G.ZENO_KEY then key = _G.ZENO_KEY end
    if getgenv then
        if getgenv().script_key then key = getgenv().script_key end
        if getgenv().ZENO_KEY then key = getgenv().ZENO_KEY end
    end
    if script_key then key = script_key end
    if ZENO_KEY then key = ZENO_KEY end
end)

lbl.Text = "Cle detectee: "..tostring(key).."\nChargement Firebase..."
task.wait(0.5)

if not key or key == "" then
    lbl.TextColor3 = Color3.fromRGB(255,50,50)
    lbl.Text = "ERREUR: Aucune cle !\n\nTu dois faire:\nscript_key = \"ZENO-TEST-123\"\nAVANT le loadstring"
    return
end

local url = "https://zeno-finder-default-rtdb.europe-west1.firebasedatabase.app/licenses/"..tostring(key)..".json"

local ok, body = pcall(function()
    -- Delta utilise parfois http_request
    if http_request then
        local res = http_request({Url=url, Method="GET"})
        return res.Body
    elseif request then
        local res = request({Url=url, Method="GET"})
        return res.Body
    else
        return game:HttpGet(url)
    end
end)

if not ok then
    lbl.TextColor3 = Color3.fromRGB(255,50,50)
    lbl.Text = "ERREUR Http:\n"..tostring(body).."\n\nTon executor bloque Http"
    return
end

lbl.Text = "Reponse Firebase:\n"..body:sub(1,200)
task.wait(1)

if body == "null" or body == "" then
    lbl.TextColor3 = Color3.fromRGB(255,50,50)
    lbl.Text = "Cle invalide !\n"..tostring(key).."\n\nCette cle n'existe pas dans Firebase\nVa verifier licenses/"..tostring(key)
    return
end

local ok2, data = pcall(function() return HttpService:JSONDecode(body) end)
if not ok2 then
    lbl.TextColor3 = Color3.fromRGB(255,50,50)
    lbl.Text = "Erreur JSON:\n"..body:sub(1,200)
    return
end

if data.active == false then
    lbl.TextColor3 = Color3.fromRGB(255,50,50)
    lbl.Text = "Cle bannie !"
    return
end

lbl.TextColor3 = Color3.fromRGB(0,255,100)
lbl.Text = "ACCES OK !\nCle: "..tostring(key).."\nChargement HUB..."
task.wait(1)
gui:Destroy()

-- CHARGE LE HUB
local hubUrl = "https://raw.githubusercontent.com/ZENOHUBPAID/loader.lua/refs/heads/main/HUB_PRIVE.txt"
local ok3, hubCode = pcall(function()
    if http_request then
        return http_request({Url=hubUrl, Method="GET"}).Body
    elseif request then
        return request({Url=hubUrl, Method="GET"}).Body
    else
        return game:HttpGet(hubUrl)
    end
end)

if not ok3 then
    warn("Erreur HUB: "..tostring(hubCode))
    -- Affiche quand meme l'erreur avec gethui
    local g2 = Instance.new("ScreenGui", getGuiParent())
    local l2 = Instance.new("TextLabel", g2)
    l2.Size = UDim2.new(0,400,0,100)
    l2.Position = UDim2.new(0.5,-200,0.5,-50)
    l2.Text = "Erreur HUB: "..tostring(hubCode)
    l2.BackgroundColor3 = Color3.fromRGB(50,0,0)
    l2.TextColor3 = Color3.fromRGB(255,255,255)
    return
end

local f, err = loadstring(hubCode)
if f then f() else 
    local g2 = Instance.new("ScreenGui", getGuiParent())
    local l2 = Instance.new("TextLabel", g2)
    l2.Size = UDim2.new(0,400,0,100)
    l2.Position = UDim2.new(0.5,-200,0.5,-50)
    l2.Text = "Erreur loadstring HUB: "..tostring(err)
    l2.BackgroundColor3 = Color3.fromRGB(50,0,0)
    l2.TextColor3 = Color3.fromRGB(255,255,255)
end
