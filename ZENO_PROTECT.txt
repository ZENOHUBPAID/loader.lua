local Players = game:GetService("Players")

local localPlayer = Players.LocalPlayer

local FIREBASE_URL = "https://zeno-finder-default-rtdb.europe-west1.firebasedatabase.app/"
local HUB_URL = "https://raw.githubusercontent.com/ZENOHUBPAID/loader.lua/refs/heads/main/HUB_PRIVE.txt"

local function httpRequest(url)
    print("[ZENO DEBUG] Requesting: "..url)
    local ok, res = pcall(function()
        if syn and syn.request then 
            print("[ZENO DEBUG] Using syn.request")
            local r = syn.request({Url=url,Method="GET"})
            return r.Body or r.body
        elseif http and http.request then 
            print("[ZENO DEBUG] Using http.request")
            local r = http.request({Url=url,Method="GET"})
            return r.Body or r.body
        elseif request then 
            print("[ZENO DEBUG] Using request")
            local r = request({Url=url,Method="GET"})
            return r.Body or r.body
        else 
            print("[ZENO DEBUG] Using HttpService:GetAsync")
            return HttpService:GetAsync(url) 
        end
    end)
    print("[ZENO DEBUG] ok="..tostring(ok).." len="..tostring(res and #res or 0))
    if not ok then print("[ZENO DEBUG] error: "..tostring(res)) end
    return ok and res or nil
end

local function denyAccess(reason)

        print("[ZENO DEBUG] Fallback to PlayerGui")
    end
    if parent:FindFirstChild("ZENO_DENIED") then parent.ZENO_DENIED:Destroy() end
    local gui = Instance.new("ScreenGui")
    gui.Name = "ZENO_DENIED"
    gui.ResetOnSpawn = false
    gui.Parent = parent
    
    local card = Instance.new("Frame")
    card.Size = UDim2.new(0, 420, 0, 190)

    card.BackgroundColor3 = Color3.fromRGB(15, 15, 18)
    card.BorderSizePixel = 0
    Instance.new("UICorner", card).CornerRadius = UDim.new(0, 16)
    card.Parent = gui
    
    local stroke = Instance.new("UIStroke", card)
    stroke.Color = Color3.fromRGB(255, 60, 60)
    stroke.Thickness = 2
    
    local title = Instance.new("TextLabel", card)
    title.Size = UDim2.new(1, 0, 0, 45)

-- CHECK LICENCE
local userKey = nil
pcall(function()
    if getgenv and getgenv().ZENO_KEY then userKey = getgenv().ZENO_KEY end
    if _G.ZENO_KEY then userKey = _G.ZENO_KEY end
end)

print("[ZENO DEBUG] userKey="..tostring(userKey))

if not userKey or userKey == "" then
    denyAccess("Aucune cle fournie. Faites getgenv().ZENO_KEY = \"VOTRE_CLE\" avant le loadstring")
    return
end

local url = FIREBASE_URL .. "licenses/" .. HttpService:UrlEncode(userKey) .. ".json"
local body = httpRequest(url)

print("[ZENO DEBUG] body="..tostring(body and body:sub(1,200) or "nil"))

if not body or body == "null" then
    denyAccess("Cle invalide: "..tostring(userKey).." (body="..tostring(body)..")")
    return
end

local ok, data = pcall(function() return HttpService:JSONDecode(body) end)
print("[ZENO DEBUG] JSON ok="..tostring(ok))
if not ok or not data then
    denyAccess("Erreur verification licence JSON")
    return
end
if data.active == false then denyAccess("Cle bannie") return end
if data.expire and data.expire < os.time() then denyAccess("Cle expiree") return end

print("[ZENO] Acces autorise - chargement... data.active="..tostring(data.active))

-- CHARGE LE VRAI HUB
local hubCode = httpRequest(HUB_URL)
print("[ZENO DEBUG] hubCode len="..tostring(hubCode and #hubCode or 0))
if hubCode and #hubCode > 100 then
    local f, err = loadstring(hubCode)
    if f then
        print("[ZENO DEBUG] loadstring OK, executing hub...")
        f()
    else
        warn("[ZENO] Erreur hub: "..tostring(err))
    end
else
    warn("[ZENO] Hub prive introuvable a "..tostring(HUB_URL).." len="..tostring(hubCode and #hubCode or 0))
end
